import { NextFunction, Response } from "express";
import { PromotionExpiredError } from "../../../errors/promotion-expired.error";
import { PromotionNotFoundError } from "../../../errors/promotion-not-found.error";
import { UserNotFoundError } from "../../../errors/user-not-found.error";
import { VenueNotFoundError } from "../../../errors/venue-not-found.error";
import { TypedRequest } from "../../../utils/typed-request";
import { AddGoogleCardDTO } from "./google.dto";
import googleSrv from "./google.service";

export const addGoogleCard = async (req: TypedRequest<AddGoogleCardDTO>, res: Response, next: NextFunction) => {
  try {
    const user = await googleSrv.getUserById(req.body.userId);
    if (!user || user.anonymizedAt != null) throw new UserNotFoundError();

    const venue = await googleSrv.getVenueById(req.body.venueId);
    if (!venue || venue.archivedAt != null) throw new VenueNotFoundError();

    const promotion = await googleSrv.getActivePromotion(req.body.venueId);
    if (!promotion) throw new PromotionNotFoundError();

    const now = new Date();
    if (promotion.startsAt > now || (promotion.endsAt !== null && promotion.endsAt < now))
      throw new PromotionExpiredError();

    const result = await googleSrv.addOrGetCard(req.body.userId, req.body.venueId, promotion.id);
    if (!result || !result.card) {
      throw new Error("Impossibile creare o recuperare la card.");
    }
    const { card, created } = result;

    const loyaltyClass = googleSrv.buildLoyaltyClass(venue);
    const loyaltyObject = googleSrv.buildLoyaltyObject(card, promotion, loyaltyClass.id);

    
    const saveUrl = googleSrv.createSaveUrl(loyaltyClass, loyaltyObject);

    res.status(created ? 201 : 200).json({
      saveUrl,
      publicCode: card.publicCode,
    });
  } catch (err) {
    next(err);
  }
};

import { and, eq, gt, isNull, lte, or } from "drizzle-orm";
import * as jwt from "jsonwebtoken";
import { customAlphabet } from "nanoid";
import { randomUUID } from "node:crypto";
import fs from "node:fs";
import path from "node:path";
import { db } from "../../../db";
import { cards, promotions, users, venues, walletProvider, walletStatus } from "../../../db/schema";
import { Card } from "../../../utils/types/card.type";
import { Promotion } from "../../../utils/types/promotion.type";
import { User } from "../../../utils/types/users.type";
import { Venue } from "../../../utils/types/venue.type";
import { brandingSchema } from "../../../utils/schemas/branding.schema";

export class GoogleService {
  private publicCodeGenerator = customAlphabet("23456789ABCDEFGHJKMNPQRSTUVWXYZ", 8);

  private generatePublicCode(): string {
    return this.publicCodeGenerator();
  }

  getPrivateKey(): string {
    const keyPath = process.env.WALLET_SERVICE_ACCOUNT_KEY_PATH;
    if (!keyPath) {
      throw new Error("WALLET_SERVICE_ACCOUNT_KEY_PATH non è definito nel file .env");
    }

    // Risolve il percorso relativo a partire dalla root del progetto
    const absolutePath = path.resolve(process.cwd(), keyPath);

    if (!fs.existsSync(absolutePath)) {
      throw new Error(`File delle credenziali Google non trovato al percorso: ${absolutePath}`);
    }

    const rawData = fs.readFileSync(absolutePath, "utf-8");
    const serviceAccount = JSON.parse(rawData);

    if (!serviceAccount.private_key) {
      throw new Error("Il file JSON non contiene il campo 'private_key'.");
    }

    return serviceAccount.private_key;
  }

  async getUserById(userId: string): Promise<User | null> {
    const [user] = await db.select().from(users).where(eq(users.id, userId)).limit(1);
    return user ?? null;
  }

  async getVenueById(venueId: string): Promise<Venue | null> {
    const [row] = await db.select().from(venues).where(eq(venues.id, venueId)).limit(1);
    if (!row) return null;

    return { ...row, branding: brandingSchema.parse(row.branding) };
  }

  async getActivePromotion(venueId: string): Promise<Promotion | null> {
    const now = new Date();
    const [promotion] = await db
      .select()
      .from(promotions)
      .where(
        and(
          eq(promotions.venueId, venueId),
          lte(promotions.startsAt, now),
          or(isNull(promotions.endsAt), gt(promotions.endsAt, now)),
        ),
      )
      .limit(1);

    return promotion ?? null;
  }

  async addOrGetCard(
    userId: string,
    venueId: string,
    promotionId: string,
  ): Promise<{ card: Card; created: boolean } | null> {
    const issuerId = process.env.WALLET_ISSUER_ID;
    if (!issuerId) return null;

    const id = randomUUID();
    const walletObjectId = `${issuerId}.${id}`;
    const publicCode = this.generatePublicCode();

    const [card] = await db
      .insert(cards)
      .values({
        id,
        userId,
        venueId,
        promotionId,
        stampsCount: 0,
        publicCode,
        walletProvider: walletProvider.enumValues[0], // "google"
        walletStatus: walletStatus.enumValues[0], // "not_added"
        walletObjectId,
        createdAt: new Date(),
      })
      .onConflictDoNothing({ target: [cards.userId, cards.venueId] })
      .returning();

    if (card) {
      return { card, created: true };
    }

    const [existingCard] = await db
      .select()
      .from(cards)
      .where(and(eq(cards.userId, userId), eq(cards.venueId, venueId)))
      .limit(1);

    return { card: existingCard, created: false };
  }

  buildLoyaltyClass(venue: Venue) {
    const issuerId = process.env.WALLET_ISSUER_ID;
    if (!issuerId) {
      throw new Error("WALLET_ISSUER_ID non è configurato.");
    }

    const branding = venue.branding;

    // ID univoco per il modello di questo locale
    const classId = `${issuerId}.${venue.id}`;

    return {
      id: classId,
      issuerName: venue.name,
      programName: venue.name,

      // Logo del locale (Google richiede un URL HTTPS valido)
      programLogo: {
        sourceUri: {
          uri: branding.logoUrl,
        },
        contentDescription: {
          defaultValue: {
            language: "it-IT",
            value: `Logo ${venue.name}`,
          },
        },
      },

      ...(branding.bannerUrl && {
        heroImage: { sourceUri: { uri: branding.bannerUrl } },
      }),

      hexBackgroundColor: branding.backgroundColor,

      // Stato di revisione: 'UNDER_REVIEW' durante i test, 'APPROVED' in produzione
      reviewStatus: "UNDER_REVIEW",
    };
  }

  buildLoyaltyObject(card: Card, promotion: Promotion, classId: string) {
    const config = (promotion.config as { stampsRequired?: number; reward?: string }) || {};

    return {
      id: card.walletObjectId, // L'ID univoco dell'oggetto (es. ISSUER_ID.uuid_card)
      classId,
      state: "ACTIVE",

      // QR Code basato sul publicCode della card (per scansione o identificazione manuale)
      barcode: {
        type: "QR_CODE",
        value: card.publicCode,
        alternateText: card.publicCode,
      },

      // Contatore principale dei timbri mostrato in primo piano nell'app Google Wallet
      loyaltyPoints: {
        label: "Timbri",
        balance: {
          int: card.stampsCount,
        },
        localizedLabel: {
          defaultValue: {
            language: "it-IT",
            value: "Timbri",
          },
        },
      },

      // Moduli di testo dinamici con l'avanzamento dell'utente
      textModulesData: [
        {
          header: "PREMIO",
          body: config.reward,
          id: "reward_info",
        },
        {
          header: "MECANISMO TIMBRI",
          body: `Raccogli ${config.stampsRequired} timbri per riscattare il premio.`,
          id: "stamps_info",
        },
        {
          header: "PROGRESSO",
          body: `${card.stampsCount}/${config.stampsRequired} timbri raccolti`,
          id: "user_progress",
        },
      ],
    };
  }

  createSaveUrl(loyaltyClass: object, loyaltyObject: object): string {
    const serviceAccountEmail = process.env.WALLET_SERVICE_ACCOUNT_EMAIL;
    if (!serviceAccountEmail) {
      throw new Error("WALLET_SERVICE_ACCOUNT_EMAIL non è configurato.");
    }

    // Carica la chiave dal file JSON
    const privateKey = this.getPrivateKey();

    const claims = {
      iss: serviceAccountEmail,
      aud: "google",
      origins: [process.env.APP_ORIGIN || "http://localhost:3000"],
      typ: "savetowallet",
      payload: {
        loyaltyClasses: [loyaltyClass],
        loyaltyObjects: [loyaltyObject],
      },
    };

    const token = jwt.sign(claims, privateKey, { algorithm: "RS256" });

    return `https://pay.google.com/gp/v/save/${token}`;
  }
}

export default new GoogleService();

  clientEmail: string;
  privateKey: string;
}

export function loadServiceAccount(keyPath: string, expectedEmail: string): ServiceAccount {
  const absolute = path.resolve(keyPath);

  let raw: unknown;
  try {
    raw = JSON.parse(readFileSync(absolute, "utf8"));
  } catch {
    throw new Error(`Impossibile leggere la chiave del service account in: ${absolute}`);
  }

  const parsed = keySchema.safeParse(raw);
  if (!parsed.success) {
    throw new Error("Il file della chiave non sembra un JSON di service account Google valido");
  }

  if (parsed.data.client_email !== expectedEmail) {
    throw new Error("WALLET_SERVICE_ACCOUNT_EMAIL nel .env non coincide con client_email del file della chiave");
  }

  return { clientEmail: parsed.data.client_email, privateKey: parsed.data.private_key };
}

export class ValidationError extends Error {
  originalErrors: OriginalValidationError[];

  constructor(errors: OriginalValidationError[]) {
    super();
    this.name = 'ValidationError';
    this.message = errors.map(error => {
        return Object.values(error.constraints!).join(', ');
      }).join(', ');
    this.originalErrors = errors;
  }
}

export const validationHandler = (err: Error, req: Request, res: Response, next: NextFunction) => {
  if (err instanceof ValidationError) {
    res.status(400);
    res.json({
      error: err.name,
      message: err.message,
      details: err.originalErrors.map(err => ({
        property: err.property,
        value: err.value,
        constraints: err.constraints
      }))
    });
  } else {
    next(err);
  }
}

import { Router } from "express";
import { createHash, randomUUID, timingSafeEqual } from "node:crypto";
import { z } from "zod";
import type { WalletProvider } from "../modules/wallet/walletProvider.js";

const sha = (v: string) => createHash("sha256").update(v).digest();

const querySchema = z
  .object({
    name: z.string().trim().min(1).max(40).default("Cliente Test"),
    venue: z
      .string()
      .trim()
      .min(1)
      .max(40)
      .regex(/^[A-Za-z0-9_-]+$/)
      .default("demo"),
    venueName: z.string().trim().min(1).max(40).default("Bar Demo"),
    stamps: z.coerce.number().int().min(0).max(50).default(0),
    required: z.coerce.number().int().min(1).max(50).default(10),
    format: z.enum(["redirect", "json"]).default("redirect"),
  })
  .refine((q) => q.stamps <= q.required, { message: "stamps non può superare required" });

export interface DevWalletDeps {
  devSecret: string;
  publicBaseUrl: string;
  walletProvider: WalletProvider;
}

export function createDevWalletRouter(deps: DevWalletDeps): Router {
  const router = Router();
  const expected = sha(deps.devSecret);

  router.get("/wallet/save-link", async (req, res, next) => {
    try {
      const provided = req.get("x-dev-secret") ?? (typeof req.query.key === "string" ? req.query.key : "");

      if (!timingSafeEqual(sha(provided), expected)) {
        res.status(401).json({ error: "unauthorized" });
        return;
      }

      if (!deps.publicBaseUrl) {
        res.status(503).json({
          error: "PUBLIC_BASE_URL vuoto: avvia il tunnel, copia l'URL nel .env e riavvia il server",
        });
        return;
      }

      const parsed = querySchema.safeParse(req.query);
      if (!parsed.success) {
        res.status(400).json({
          error: "invalid_query",
          issues: parsed.error.issues.map((i) => ({ path: i.path.join("."), message: i.message })),
        });
        return;
      }

      const q = parsed.data;
      const cardId = randomUUID();

      const link = await deps.walletProvider.createSaveLink({
        cardId,
        venueId: q.venue,
        venueName: q.venueName,
        holderName: q.name,
        stamps: q.stamps,
        stampsRequired: q.required,
      });

      console.log(`[dev] pass generato: card=${cardId} venue=${q.venue}`);
      res.setHeader("Cache-Control", "no-store");

      if (q.format === "json") {
        res.json({ cardId, link });
        return;
      }
      res.redirect(302, link);
    } catch (err) {
      next(err);
    }
  });

  return router;
}

=== src/utils/validation-middleware.ts ===
import { plainToClass } from "class-transformer";
import { NextFunction, Response } from "express";
import { validate as classValidate } from "class-validator";
import { TypedRequest } from "./typed-request";
import { ValidationError } from "../errors/validation-error";

function validateFn<T extends object>(dtoClass: new() => T, origin: 'body')
  : (req: TypedRequest<T, unknown, unknown>, res: Response, next: NextFunction) => Promise<void>;
function validateFn<T extends object>(dtoClass: new() => T, origin: 'query')
  : (req: TypedRequest<unknown, T, unknown>, res: Response, next: NextFunction) => Promise<void>;
function validateFn<T extends object>(dtoClass: new() => T, origin: 'params')
  : (req: TypedRequest<unknown, unknown, T>, res: Response, next: NextFunction) => Promise<void>;
function validateFn<T extends object>(dtoClass: new() => T, origin: 'body' | 'query' | 'params') {
  return async function(req: TypedRequest<any, any, any>, res: Response, next: NextFunction) {
    if (!req[origin]) {
      throw new Error(`Missing ${origin}`);
    }
    const data = plainToClass(dtoClass, req[origin]);
    const errors = await classValidate(data);
    if (errors.length === 0) {
      // fix for express > 5: makes req.query writable
      if (origin === 'query') {
        Object.defineProperty(
          req,
          'query',
          { ...Object.getOwnPropertyDescriptor(req, 'query'),
            value: req.query,
            writable: true
          });
      }
      req[origin] = data;
      next();
    } else {
      next(new ValidationError(errors));
    }
  }
}

export const validate = validateFn;
